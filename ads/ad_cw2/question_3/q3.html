# -----------------------------
# Bootstrap particle filter
# returns log-likelihood estimate
# -----------------------------
def PF(y, sigma2, beta2, rho, num_particles=1000, rng=rng):

    N = int(num_particles)
    T = len(y)

    if sigma2 <= 0 or beta2 <= 0 or not (-1 < rho < 1):
        return -np.inf, None, None

    particles = np.zeros((N, T))
    weights = np.zeros((N, T))
    loglik_hat = 0.0

    # X0 ~ N(0, 0.01)  <-- follow the question
    particles[:, 0] = rng.normal(0.0, np.sqrt(0.01), size=N)

    # Y0 | X0 ~ N(0, beta^2 exp(X0))
    logw0 = (
        -0.5 * np.log(2 * np.pi * beta2 * np.exp(particles[:, 0]))
        -0.5 * (y[0] ** 2) / (beta2 * np.exp(particles[:, 0]))
    )

    m0 = np.max(logw0)
    w0_unnorm = np.exp(logw0 - m0)
    mean_w0 = np.mean(w0_unnorm)

    if mean_w0 <= 0 or not np.isfinite(mean_w0):
        return -np.inf, None, None

    loglik_hat += m0 + np.log(mean_w0)

    w = w0_unnorm / np.sum(w0_unnorm)
    weights[:, 0] = w

    idx = rng.choice(N, size=N, p=w)
    particles[:, 0] = particles[idx, 0]
    weights[:, 0] = 1.0 / N

    for t in range(1, T):
        # state transition: X_t = rho X_{t-1} + sigma V_t
        particles[:, t] = rho * particles[:, t-1] + rng.normal(0.0, np.sqrt(sigma2), size=N)

        # Y_t | X_t ~ N(0, beta^2 exp(X_t))
        logwt = (
            -0.5 * np.log(2 * np.pi * beta2 * np.exp(particles[:, t]))
            -0.5 * (y[t] ** 2) / (beta2 * np.exp(particles[:, t]))
        )

        mt = np.max(logwt)
        wt_unnorm = np.exp(logwt - mt)
        mean_wt = np.mean(wt_unnorm)

        if mean_wt <= 0 or not np.isfinite(mean_wt):
            return -np.inf, None, None

        loglik_hat += mt + np.log(mean_wt)

        w = wt_unnorm / np.sum(wt_unnorm)
        weights[:, t] = w

        idx = rng.choice(N, size=N, p=w)
        particles[:, t] = particles[idx, t]
        weights[:, t] = 1.0 / N

    return loglik_hat, particles, weights

from scipy.stats import invgamma

def log_prior(sigma2, beta2, rho):
    if sigma2 <= 0 or beta2 <= 0 or not (-1 <= rho <= 1):
        return -np.inf

    lp_sigma = invgamma.logpdf(sigma2, a=1, scale=1)
    lp_beta  = invgamma.logpdf(beta2, a=1, scale=1)
    lp_rho   = -np.log(2.0)   # Uniform[-1,1]

    return lp_sigma + lp_beta + lp_rho


# -----------------------------
# PMMH / PMCMC
# returns sigma2_path, beta2_path, rho_path
# -----------------------------
def PMCMC(
    y,
    sigma2_init,
    beta2_init,
    rho_init,
    num_particles=1000,
    num_iterations=2000,
    step_sigma=0.15,
    step_beta=0.15,
    step_rho=0.15,
    rng=rng
):

    sigma2_path = np.zeros(num_iterations)
    beta2_path  = np.zeros(num_iterations)
    rho_path    = np.zeros(num_iterations)
    accept      = np.zeros(num_iterations, dtype=int)

    sigma2_path[0] = sigma2_init
    beta2_path[0]  = beta2_init
    rho_path[0]    = rho_init

    sigma2_curr = sigma2_init
    beta2_curr  = beta2_init
    rho_curr    = rho_init

    logprior_curr = log_prior(sigma2_curr, beta2_curr, rho_curr)
    loglik_curr, _, _ = PF(
        y, sigma2_curr, beta2_curr, rho_curr,
        num_particles=num_particles, rng=rng
    )

    for i in range(1, num_iterations):
        # propose in transformed space
        log_sigma2_prop = np.log(sigma2_curr) + rng.normal(0.0, step_sigma)
        log_beta2_prop  = np.log(beta2_curr) + rng.normal(0.0, step_beta)

        eta_curr = np.arctanh(np.clip(rho_curr, -0.999999, 0.999999))
        eta_prop = eta_curr + rng.normal(0.0, step_rho)

        sigma2_prop = np.exp(log_sigma2_prop)
        beta2_prop  = np.exp(log_beta2_prop)
        rho_prop    = np.tanh(eta_prop)

        logprior_prop = log_prior(sigma2_prop, beta2_prop, rho_prop)

        if np.isfinite(logprior_prop):
            loglik_prop, _, _ = PF(
                y, sigma2_prop, beta2_prop, rho_prop,
                num_particles=num_particles, rng=rng
            )

            if np.isfinite(loglik_prop):
                # Jacobian terms from transformation
                log_jac_curr = (
                    np.log(sigma2_curr)
                    + np.log(beta2_curr)
                    + np.log(1.0 - rho_curr**2)
                )
                log_jac_prop = (
                    np.log(sigma2_prop)
                    + np.log(beta2_prop)
                    + np.log(1.0 - rho_prop**2)
                )

                log_alpha = (
                    (loglik_prop - loglik_curr)
                    + (logprior_prop - logprior_curr)
                    + (log_jac_prop - log_jac_curr)
                )

                if np.log(rng.uniform()) < min(0.0, log_alpha):
                    sigma2_curr = sigma2_prop
                    beta2_curr  = beta2_prop
                    rho_curr    = rho_prop
                    logprior_curr = logprior_prop
                    loglik_curr = loglik_prop
                    accept[i] = 1

        sigma2_path[i] = sigma2_curr
        beta2_path[i]  = beta2_curr
        rho_path[i]    = rho_curr

    return sigma2_path, beta2_path, rho_path

    axes[2].set_ylabel(r'$\rho$')

def acf(x, max_lag=50):
    x = np.asarray(x)
    x = x - np.mean(x)
    n = len(x)

    var = np.dot(x, x) / n
    vals = [1.0]

    for lag in range(1, max_lag + 1):
        val = np.dot(x[:-lag], x[lag:]) / ((n - lag) * var)
        vals.append(val)

    return np.array(vals)
